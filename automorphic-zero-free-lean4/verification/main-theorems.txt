ERH.BootstrapElliptic.FactorEstimates (F : ℂ → ℂ) : Prop
def ERH.BootstrapElliptic.FactorEstimates : (ℂ → ℂ) → Prop :=
fun F =>
  ∀ (η : ℝ),
    0 < η →
      ∃ H,
        ∀ (β t : ℝ),
          H ≤ Real.log |t| →
            0 < β →
              β < 1 →
                F { re := β, im := t } = 0 →
                  Real.log |t| * (1 - β) ≤ ERH.BootstrapElliptic.W →
                    ∃ s a11 a51 a22 a82,
                      -η ≤ s + a11 + 13978 / 1000000 * a51 + 143554 / 1000000 * a22 + 944 / 1000000 * a82 ∧
                        s ≤
                            ERH.BootstrapElliptic.p *
                                ERH.BootstrapElliptic.lap (-ERH.BootstrapElliptic.d / ERH.BootstrapElliptic.W) +
                              η ∧
                          a11 ≤
                              ERH.BootstrapElliptic.W / 3 + η -
                                ERH.BootstrapElliptic.lap
                                  (Real.log |t| * (1 - β) / ERH.BootstrapElliptic.W -
                                    ERH.BootstrapElliptic.d / ERH.BootstrapElliptic.W) ∧
                            a51 ≤ ERH.BootstrapElliptic.W / 3 + η ∧
                              a22 ≤ ERH.BootstrapElliptic.W * (139 / 186) + η ∧
                                a82 ≤ ERH.BootstrapElliptic.W * (139 / 186) + η
